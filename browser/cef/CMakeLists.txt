cmake_minimum_required(VERSION 3.21)
project(nida_browser_host LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Nida CEF host requires Windows x64")
endif()
if(NOT DEFINED CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to an extracted Windows x64 CEF binary distribution")
endif()
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox is required" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
# CEF 152 uses a sandbox bootstrap EXE loading our client DLL.
add_library(nida-browser-host SHARED main.cpp)
SET_LIBRARY_TARGET_PROPERTIES(nida-browser-host)
target_compile_features(nida-browser-host PRIVATE cxx_std_20)
target_include_directories(nida-browser-host PRIVATE "${CEF_ROOT}")
target_link_libraries(nida-browser-host PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
# Keep helper and all resource-only dependencies together, independent of CWD.
install(FILES "$<TARGET_FILE:nida-browser-host>" DESTINATION browser)
install(FILES "${CEF_ROOT}/Release/bootstrap.exe" DESTINATION browser RENAME nida-browser-host.exe)
install(DIRECTORY "${CEF_ROOT}/Release/" DESTINATION browser
  FILES_MATCHING PATTERN "*.dll" PATTERN "*.bin" PATTERN "*.json")
install(DIRECTORY "${CEF_ROOT}/Resources/" DESTINATION browser)
install(FILES "${CEF_ROOT}/LICENSE.txt" DESTINATION browser RENAME CEF-LICENSE.txt)
install(FILES "${CEF_ROOT}/include/cef_version.h" DESTINATION browser)
